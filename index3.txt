    // FAQ 토글
    document.querySelectorAll('.faq-q').forEach(btn => {
      btn.addEventListener('click', () => {
        const targetId = btn.getAttribute('data-target');
        const ans = document.getElementById(targetId);
        const expanded = btn.getAttribute('aria-expanded') === 'true';
        btn.setAttribute('aria-expanded', String(!expanded));
        ans.style.display = expanded ? 'none' : 'block';
      });
    });

    // 요금표 버튼 상호작용 예시: 동일 높이 유지
    function togglePlan(planName){
      console.log(planName + " 플랜 클릭됨");
      const msg = document.createElement('div');
      msg.textContent = planName + " 플랜이 선택되었습니다.";
      msg.style.position = 'fixed';
      msg.style.bottom = '20px';
      msg.style.left = '50%';
      msg.style.transform = 'translateX(-50%)';
      msg.style.background = '#a2d149';
      msg.style.color = '#111';
      msg.style.fontWeight = 'bold';
      msg.style.padding = '12px 20px';
      msg.style.borderRadius = '8px';
      msg.style.boxShadow = '0 6px 14px rgba(0,0,0,.3)';
      msg.style.zIndex = 9999;
      document.body.appendChild(msg);
      setTimeout(()=>{ msg.remove(); }, 1500);
    }

    // 신청 버튼 클릭 시 데모 안내 표시

    const demoBox = document.getElementById('demoBox');
    applyBtn.addEventListener('click', ()=>{
      demoBox.style.display = (demoBox.style.display === 'block') ? 'none' : 'block';
    });
